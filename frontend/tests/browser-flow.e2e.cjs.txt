const { chromium } = require('/opt/npm-tools/node_modules/playwright')
;(async () => {
  const b = await chromium.launch(); const ctx = await b.newContext({ viewport: { width: 390, height: 844 }, deviceScaleFactor: 1 })
  const p = await ctx.newPage(); const errs = []
  p.on('pageerror', (e) => errs.push('PAGEERR ' + e.message)); p.on('console', (m) => { if (m.type() === 'error') errs.push('CONSOLE ' + m.text()) })
  p.on('dialog', (d) => d.accept())
  const log = (...a) => console.log(...a)
  await p.goto('http://localhost:5199/'); await p.waitForSelector('.period__tabs')
  log('tabs:', await p.locator('.period__tab').allTextContents())
  log('folders today:', await p.locator('.fc__name').allTextContents())
  await p.screenshot({ path: `/tmp/claude-0/-home-claude/4bc65422-0490-5cf6-9c8b-b1da767d72f9/scratchpad/s1.png` })
  log('overflow x:', await p.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth))
  // page-level checklist (no folder)
  await p.getByRole('button', { name: '+ Add checklist' }).first().click()
  await p.fill('input[aria-label="Amount in rupees"]', '30'); await p.fill('input[aria-label="Checklist name (optional)"]', 'Milk')
  await p.getByRole('button', { name: 'Create Checklist' }).click(); await p.waitForSelector('.cr__name:text("Milk")')
  await p.locator('.cr', { hasText: 'Milk' }).locator('.cr__rec').click(); await p.waitForTimeout(300)
  log('page checklist Milk records:', await p.locator('.cr', { hasText: 'Milk' }).locator('.cr__count b').textContent(), '(expect 1)')
  // open first folder inline
  await p.locator('.fc__head').first().click()
  await p.waitForSelector('.fc__inside')
  log('open ok; checklists:', await p.locator('.cr__name').allTextContents())
  // add checklist Tea 15
  await p.getByRole('button', { name: '+ Add checklist' }).nth(1).click()
  await p.fill('input[aria-label="Amount in rupees"]', '15')
  await p.fill('input[aria-label="Checklist name (optional)"]', 'Tea15')
  await p.getByRole('button', { name: 'Create Checklist' }).click()
  await p.waitForSelector('.cr__name:text("Tea15")')
  const row = p.locator('.cr', { hasText: 'Tea15' })
  for (let i = 0; i < 5; i++) await row.locator('.cr__rec').click()
  await row.locator('.cr__other').click(); await p.fill('.cr__input', '20'); await p.keyboard.press('Enter')
  await p.waitForTimeout(500)
  log('Tea15 records box:', await row.locator('.cr__count b').textContent(), '(expect 6)')
  await p.screenshot({ path: `/tmp/claude-0/-home-claude/4bc65422-0490-5cf6-9c8b-b1da767d72f9/scratchpad/s2.png`, fullPage: true })
  // add inner folder (auto name) then add item
  await p.getByRole('button', { name: '+ Add folder' }).first().click(); await p.getByRole('button', { name: 'Create Folder' }).click()
  await p.waitForTimeout(400)
  log('names:', await p.locator('.fc__name').allTextContents())
  const inner = p.locator('.fc--inner').first()
  log('inner records before:', await inner.locator('.fc__count b').first().textContent())
  await inner.locator('.fc__add').first().click(); await p.waitForTimeout(300)
  log('inner records after +Add item:', await inner.locator('.fc__count b').first().textContent(), '(expect 2)')
  // twin: + Edit on closed inner folder
  const n0 = await p.locator('.fc--inner').count()
  await inner.locator('.fc__edit').first().click(); await p.waitForTimeout(300)
  log('inner folders before/after twin:', n0, await p.locator('.fc--inner').count())
  // saving area card -> page
  await p.locator('.saving-area').last().scrollIntoViewIfNeeded()
  await p.screenshot({ path: `/tmp/claude-0/-home-claude/4bc65422-0490-5cf6-9c8b-b1da767d72f9/scratchpad/s3.png`, fullPage: true })
  log('overflow x (open card):', await p.evaluate(() => document.documentElement.scrollWidth - document.documentElement.clientWidth))
  await p.locator('.saving-area--link').first().click(); await p.waitForSelector('.sa-page')
  log('saving page title:', await p.locator('.sa-head h1').textContent())
  await p.screenshot({ path: `/tmp/claude-0/-home-claude/4bc65422-0490-5cf6-9c8b-b1da767d72f9/scratchpad/s4.png` })
  await p.locator('.sa-head button').first().click(); await p.waitForSelector('.period__tabs')
  // reload persistence
  await p.reload(); await p.waitForSelector('.period__tabs')
  await p.locator('.fc__head').first().click(); await p.waitForSelector('.fc__inside')
  log('after reload Tea15 records:', await p.locator('.cr', { hasText: 'Tea15' }).locator('.cr__count b').textContent(), 'checklists:', await p.locator('.cr__name').allTextContents())
  // history shows 6 records
  await p.getByRole('button', { name: /History/ }).click(); await p.waitForTimeout(500)
  log('history rows with Tea15:', await p.locator('text=Tea15').count())
  log('ERRORS:', errs.length ? errs : 'none')
  await b.close()
})().catch((e) => { console.log('FAIL', e.message); process.exit(1) })
